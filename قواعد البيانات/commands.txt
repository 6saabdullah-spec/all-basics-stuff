Create Table college(          *CREATED DATABASE WITH NAME*
ID INT(10) Primary key,        *PRIMARY KEY DOSENT REPEAT AND*
fname Text(50) Not Null,       *NOT NULL MEANS DOSENT LEFT EMPTY*
lname Text(50) Not Null,
Email Text(50) Unique,         *UNIQUE MEANS EMAIL CAN NOT BE USED WITH ANOTHER USER*
age INT(2) check (age >= 18)   *CHECK IS MAKING SURE THE USER IS HIGHER THAN 18 YEAR OLD ETHIER IT DOSENT ACCEPT*
);

Create Table course(
ID INT(10) Primary key,
cname Text(50) Not Null,
SID INT(10),
foreign key (SID) references college(ID) *FOREIGN MEANS THAT THE KEY IS BACK TO THE MAIN KEY TO COLLAGE*
);
